using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[Logger]
public interface IAuditServiceLogs
{
	[Info("Audit: {Action}")]
	void Audit(string action);

}

public class AuditService
{
	readonly IAuditServiceLogs _logger;

	public AuditService(IAuditServiceLogs logger) => _logger = logger;

	public void LogAudit(string action)
	{
		_logger.Audit(action);
	}
}
