using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[Logger]
public interface IWeatherServiceLogs
{
	[Info("Fetching weather for {City}")]
	void FetchingWeatherFor(string city);

}

public class WeatherService
{
	readonly IWeatherServiceLogs _logger;

	public WeatherService(IWeatherServiceLogs logger) => _logger = logger;

	public void GetWeather(string city)
	{
		_logger.FetchingWeatherFor(city);
	}
}
[Logger]
public interface IOrderServiceLogs
{
	[Info("Order {OrderId} placed")]
	void OrderPlaced(int orderId);

}

public class OrderService
{
	readonly IOrderServiceLogs _logger;

	public OrderService(IOrderServiceLogs logger) => _logger = logger;

	public void PlaceOrder(int orderId)
	{
		_logger.OrderPlaced(orderId);
	}
}
