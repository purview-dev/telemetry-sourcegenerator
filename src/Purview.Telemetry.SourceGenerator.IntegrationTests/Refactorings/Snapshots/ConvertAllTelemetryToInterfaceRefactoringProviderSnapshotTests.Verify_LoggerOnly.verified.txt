using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[Logger]
public interface IAuditServiceTelemetry
{
	[Info("User {UserId} performed {Action}")]
	void UserPerformed(string userId, string action);

}

public class AuditService
{
	readonly IAuditServiceTelemetry _logger;

	public AuditService(IAuditServiceTelemetry logger) => _logger = logger;

	public void LogAction(string userId, string action)
	{
		_logger.UserPerformed(userId, action);
	}
}
}
