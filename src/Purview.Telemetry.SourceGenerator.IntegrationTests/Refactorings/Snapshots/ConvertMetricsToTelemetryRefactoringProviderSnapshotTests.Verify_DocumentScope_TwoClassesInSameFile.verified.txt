using System.Diagnostics.Metrics;
using Purview.Telemetry;

namespace Testing;
[Meter]
public interface IRequestMetricsMetrics
{
	[AutoCounter]
	void RequestCount();

}

public class RequestMetrics
{
	readonly IRequestMetricsMetrics _requestCount;

	public RequestMetrics(IMeterFactory meterFactory)
	{
		var meter = meterFactory.Create("RequestMetrics");
		_requestCount = meter.CreateCounter<long>("request.count");
	}

	public void RecordRequest()
	{
		_requestCount.RequestCount();
	}
}
[Meter]
public interface IErrorMetricsMetrics
{
	[AutoCounter]
	void ErrorCount();

}

public class ErrorMetrics

	{
		var meter = meterFactory.Create("ErrorMetrics");
