using System.Diagnostics.Metrics;
using Purview.Telemetry;

namespace Testing;
[Meter]
public interface IStorageMetricsMetrics
{
	[AutoCounter]
	void Writes();

	[AutoCounter]
	void Reads();

}

public class StorageMetrics(IMeterFactory meterFactory)
{
	readonly IStorageMetricsMetrics _writes = meterFactory.Create("StorageMetrics").CreateCounter<long>("writes");
	readonly IStorageMetricsMetrics _reads = meterFactory.Create("StorageMetrics").CreateCounter<long>("reads");

	public void RecordWrite() => _writes.Writes();
	public void RecordRead() => _reads.Reads();
}
