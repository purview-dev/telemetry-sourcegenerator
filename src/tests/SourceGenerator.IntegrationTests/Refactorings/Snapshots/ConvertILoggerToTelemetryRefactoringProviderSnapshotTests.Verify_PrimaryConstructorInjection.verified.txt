using Microsoft.Extensions.Logging;
using Purview.Telemetry;

public interface IInventoryServiceLogs
{

	void StockUpdateBy(string sku, int delta);

}

public class InventoryService(IInventoryServiceLogs logger)
{

	{
		logger.StockUpdateBy(sku, delta);
	}
