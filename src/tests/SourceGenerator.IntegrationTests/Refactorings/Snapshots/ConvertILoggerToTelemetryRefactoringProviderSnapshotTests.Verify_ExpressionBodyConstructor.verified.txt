using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[Logger]
public interface ICacheServiceLogs
{
	[Debug("Cache evict: {Key}")]
	void CacheEvict(string key);

}

public class CacheService
{
	readonly ICacheServiceLogs _log;

	public CacheService(ICacheServiceLogs log) => _log = log;

	public void Evict(string key)
	{
		_log.CacheEvict(key);
	}
}
}
