using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[Logger]
public interface IWeatherServiceLogs
{
	[Info("Getting weather for {City} over {Days} days")]
	void GettingWeatherForOverDays(string city, int days);

}

public class WeatherService
{
	readonly IWeatherServiceLogs _logger;

	public WeatherService(IWeatherServiceLogs logger) => _logger = logger;

	public void GetWeather(string city, int days)
	{
		_logger.GettingWeatherForOverDays(city, days);
	}
}
