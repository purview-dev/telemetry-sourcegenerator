using System.Diagnostics;

namespace Testing;
[ActivitySource]
public interface IEmailServiceTracing
{
	[Activity]
	Activity? SendEmail();

}

public class EmailService
{
	readonly IEmailServiceTracing _tracer;

	public EmailService(IEmailServiceTracing tracer) => _tracer = tracer;

	public void Send(string to, string subject)
	{
		using var activity = _tracer.SendEmail();
	}
}
