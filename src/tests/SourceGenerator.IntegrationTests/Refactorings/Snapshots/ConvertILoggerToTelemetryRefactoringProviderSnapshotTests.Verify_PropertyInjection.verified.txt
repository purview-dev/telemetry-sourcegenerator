using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[Logger]
public interface INotificationServiceLogs
{
	[Info("Notifying {UserId}: {Message}")]
	void Notifying(string userId, string message);

}

public class NotificationService
{
	public INotificationServiceLogs Logger { get; set; } = null!;

	public void Notify(string userId, string message)
	{
		Logger.Notifying(userId, message);
	}
}
