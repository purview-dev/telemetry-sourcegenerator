using System;
using System.Diagnostics;
using System.Diagnostics.Metrics;
using Microsoft.Extensions.Logging;
using Purview.Telemetry;

namespace Testing;
[ActivitySource]
[Logger]
[Meter]
public interface IProductCatalogServiceTelemetry
{
	[Activity]
	Activity? Search();

	[Activity]
	Activity? IndexProduct();

	[Debug("Searching catalog: {Query} (page={PageSize})")]
	void SearchingCatalogPage(string query, int pageSize);

	[Info("Search completed in {LatencyMs}ms for {Query}")]
	void SearchCompletedInMsFor(double latencyMs, string query);

	[Info("Indexing product {ProductId}")]
	void IndexingProduct(string productId);

	[Error("Failed to index product {ProductId}")]
	void FailedToIndexProduct(System.Exception exception, string productId);

	[AutoCounter]
	void Searches();

	[Histogram]
	void SearchLatency(double value);

}

public class ProductCatalogService(
	IProductCatalogServiceTelemetry logger,
	IMeterFactory meterFactory)
{
	static readonly ActivitySource _activitySource = new("ProductCatalogService");
	readonly Counter<long> _searches = meterFactory.Create("ProductCatalogService").CreateCounter<long>("catalog-searches");
	readonly Histogram<double> _searchLatency = meterFactory.Create("ProductCatalogService").CreateHistogram<double>("search-latency-ms");

	public void Search(string query, int pageSize)
	{
		using var activity = _activitySource.StartActivity("Search", ActivityKind.Internal);
		logger.SearchingCatalogPage(query, pageSize);
		_searches.Add(1);
	}

	public void RecordSearchLatency(double latencyMs, string query)
	{
		_searchLatency.Record(latencyMs);
		logger.SearchCompletedInMsFor(latencyMs, query);
	}

	public void IndexProduct(string productId)
	{
		using var activity = _activitySource.StartActivity("IndexProduct");
		logger.IndexingProduct(productId);
	}

	public void IndexFailed(string productId, Exception ex)
	{
		logger.FailedToIndexProduct(ex, productId);
	}
}
}
